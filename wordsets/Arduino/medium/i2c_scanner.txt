#include <Wire.h>
void setup() {
	Wire.begin();
	Serial.begin(9600);
	Serial.println("Scanning I2C...");
}
void loop() {
	byte e, a;
	for (a = 1; a < 127; a++) {
		Wire.beginTransmission(a);
		e = Wire.endTransmission();
		if (e == 0) {
			Serial.print("Device at 0x");
			Serial.println(a, HEX);
		}
	}
	delay(5000);
}
